from pathlib import Path
from playwright.sync_api import sync_playwright,expect
from scripts.stage9_common import Demo

with sync_playwright() as p:
 browser=p.chromium.launch(headless=True);ctx=browser.new_context(viewport=dict(width=1366,height=900),timezone_id='Europe/Moscow');page=ctx.new_page()

 page.goto(url+'/login');page.locator('[name=username]').fill('demo-admin');page.locator('[name=password]').fill((d.root/'accounts/admin.password').read_text().strip())

 page.set_viewport_size(dict(width=390,height=844));page.reload();expect(page.locator('#effective-profile')).to_have_text('Ascon-AEAD128');assert page.evaluate('document.documentElement.scrollWidth <= innerWidth+2')
 page.screenshot(path=str(root/'device-alpha-mobile-final.png'),full_page=True)
 browser.close()
